<script>
    // --- Drag & Drop Logic ---
    const windowEl = document.getElementById('calculatorWindow');
    const dragHandle = document.getElementById('dragHandle');
    let isDragging = false;
    let initialX, initialY, initialLeft, initialTop;

    dragHandle.addEventListener('mousedown', (e) => {
        e.preventDefault();
        isDragging = true;
        initialX = e.clientX;
        initialY = e.clientY;
        initialLeft = windowEl.offsetLeft;
        initialTop = windowEl.offsetTop;
    });

    document.addEventListener('mousemove', (e) => {
        if (!isDragging) return;
        const deltaX = e.clientX - initialX;
        const deltaY = e.clientY - initialY;
        windowEl.style.left = `${initialLeft + deltaX}px`;
        windowEl.style.top = `${initialTop + deltaY}px`;
    });

    document.addEventListener('mouseup', () => {
        isDragging = false;
    });

    // Close button
    document.getElementById('closeBtn').addEventListener('click', () => {
        windowEl.remove();
    });

    // --- Calculator Logic ---
    const display = document.getElementById('display');
    
    let currentInput = '';
    let previousInput = '';
    let operation = null;
    let shouldResetDisplay = false;
    
    // Memory
    let memory = 0;

    function updateDisplay() {
        display.value = currentInput || '0';
    }

    function inputDigit(digit) {
        if (shouldResetDisplay) {
            currentInput = '';
            shouldResetDisplay = false;
        }
        if (digit === '.' && currentInput.includes('.')) return;
        if (currentInput === '0' && digit !== '.') {
            currentInput = digit;
        } else {
            currentInput += digit;
        }
        updateDisplay();
    }

    function setOperation(op) {
        if (currentInput === '') return;
        if (previousInput !== '') {
            calculate();
        }
        operation = op;
        previousInput = currentInput;
        currentInput = '';
        shouldResetDisplay = false;
    }

    function calculate() {
        let result;
        const prev = parseFloat(previousInput);
        const current = parseFloat(currentInput);

        if (isNaN(prev) || isNaN(current)) return;

        switch (operation) {
            case '+': result = prev + current; break;
            case '-': result = prev - current; break;
            case '*': result = prev * current; break;
            case '/': 
                if (current === 0) { alert('Ошибка деления на ноль'); return; }
                result = prev / current; 
                break;
            default: return;
        }

        currentInput = String(result);
        previousInput = '';
        operation = null;
        updateDisplay();
    }

    function clearAll() {
        currentInput = '0';
        previousInput = '';
        operation = null;
        shouldResetDisplay = false;
        updateDisplay();
    }

    function clearEntry() {
        currentInput = '0';
        updateDisplay();
    }

    function backspace() {
        if (shouldResetDisplay) return;
        currentInput = currentInput.toString().slice(0, -1);
        if (currentInput === '' || currentInput === '-') {
            currentInput = '0';
        }
        updateDisplay();
    }

    function toggleSign() {
        if (currentInput !== '0') {
            currentInput = String(-parseFloat(currentInput));
            updateDisplay();
        }
    }

    function sqrt() {
        const val = parseFloat(currentInput);
        if (val < 0) { alert('Нельзя извлечь корень из отрицательного числа'); return; }
        currentInput = String(Math.sqrt(val));
        updateDisplay();
    }

    function percent() {
        // Если есть предыдущее значение и операция, считаем процент от него
        if (previousInput && operation) {
            const percentValue = (parseFloat(currentInput) / 100) * parseFloat(previousInput);
            currentInput = String(percentValue);
        } else {
            currentInput = String(parseFloat(currentInput) / 100);
        }
        updateDisplay();
    }

    function inverse() {
        const val = parseFloat(currentInput);
        if (val === 0) { alert('Деление на ноль невозможно'); return; }
        currentInput = String(1 / val);
        updateDisplay();
    }

    // Memory Functions
    function memoryClear() { memory = 0; }
    function memoryRecall() { currentInput = String(memory); updateDisplay(); }
    function memoryStore() { memory = parseFloat(currentInput); }
    function memoryAdd() { 
        if (currentInput !== '') memory += parseFloat(currentInput); 
    }

    // Event Listeners for Buttons
    document.querySelectorAll('button').forEach(btn => {
        const text = btn.textContent;
        
        // Digits
        if (/^\d$/.test(text)) {
            btn.addEventListener('click', () => inputDigit(text));
        }
        
        // Operations
        if (text === '+') btn.addEventListener('click', () => setOperation('+'));
        if (text === '-') btn.addEventListener('click', () => setOperation('-'));
        if (text === '*') btn.addEventListener('click', () => setOperation('*'));
        if (text === '/') btn.addEventListener('click', () => setOperation('/'));
        
        // Equals
        if (text === '=') btn.addEventListener('click', calculate);
        
        // Clear
        if (text === 'C') btn.addEventListener('click', clearAll);
        if (text === 'CE') btn.addEventListener('click', clearEntry);
        if (text === 'Backspace') btn.addEventListener('click', backspace);
        
        // Specials
        if (text === '+/-') btn.addEventListener('click', toggleSign);
        if (text === '.') btn.addEventListener('click', () => inputDigit('.'));
        if (text === 'sqrt') btn.addEventListener('click', sqrt);
        if (text === '%') btn.addEventListener('click', percent);
        if (text === '1/x') btn.addEventListener('click', inverse);
        
        // Memory
        if (text === 'MC') btn.addEventListener('click', memoryClear);
        if (text === 'MR') btn.addEventListener('click', memoryRecall);
        if (text === 'MS') btn.addEventListener('click', memoryStore);
        if (text === 'M+') btn.addEventListener('click', memoryAdd);
    });
</script>
